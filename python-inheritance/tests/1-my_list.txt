#!/usr/bin/python3
"""Doctest for MyList class."""

MyList = __import__('1-my_list').MyList

# Test a normal list
my_list = MyList([5, 1, 3, 2, 4])
my_list.print_sorted()
print(list(my_list))

# Test empty list
empty = MyList()
empty.print_sorted()
print(list(empty))

# Test single-item list
single = MyList([42])
single.print_sorted()
print(list(single))

# Test negative and positive numbers
nums = MyList([-3, 0, 2, -1, 5])
nums.print_sorted()
print(list(nums))

# Test append method
append_test = MyList()
append_test.append(10)
append_test.append(1)
append_test.append(5)
append_test.print_sorted()
print(list(append_test))
